# --- login_method: "sfapi" (NERSC Superfacility API) ---

PATH_NERSC_PRI_KEY=/path/to/nersc_private_key.json

# --- login_method: "iriapi" (NERSC IRI API, Globus token) ---
# Your NERSC username (used for /pscratch paths)

# Token file created once with: python flows/sfapi/globus_token.py --validate-iri
# (default if unset: ~/.globus/auth_tokens.json)
PATH_GLOBUS_TOKEN_FILE=/path/to/.globus/auth_tokens.json
