## 🎯 Step-by-Step: What to Actually Do

### Step 1: Organize Your Raw Data

You mentioned you have:
- N images (e.g., `image_001.jpg`, `image_002.jpg`, ...)
- N mask images (e.g., `image_001.png`, `image_002.png`, ...)
- Mask pixel values: `0=ignore`, `1=sand`, `2=air`, `3=background`

Put them here:

```bash
mkdir -p my_dataset/raw/images
mkdir -p my_dataset/raw/masks

# Copy your files
cp /path/to/your/images/*.jpg my_dataset/raw/images/
cp /path/to/your/masks/*.png my_dataset/raw/masks/
```

### Step 2: Split Into Train/Valid/Test

```bash
mkdir -p my_dataset/train/images my_dataset/train/masks
mkdir -p my_dataset/valid/images my_dataset/valid/masks
mkdir -p my_dataset/test/images my_dataset/test/masks

# Example split: 70% train, 15% valid, 15% test
# Use a simple Python script or manually move files
```

Simple split script:

**File: `split_dataset.py`**

```python
import os
import shutil
import random
from pathlib import Path

# Configuration
raw_images = "my_dataset/raw/images"
raw_masks = "my_dataset/raw/masks"
train_split = 0.7
valid_split = 0.15
test_split = 0.15

# Get all image files
image_files = sorted([f for f in os.listdir(raw_images) if f.endswith(('.jpg', '.png'))])
random.shuffle(image_files)

# Calculate splits
n_train = int(len(image_files) * train_split)
n_valid = int(len(image_files) * valid_split)

train_files = image_files[:n_train]
valid_files = image_files[n_train:n_train + n_valid]
test_files = image_files[n_train + n_valid:]

# Copy files
for split, files in [('train', train_files), ('valid', valid_files), ('test', test_files)]:
    for img_file in files:
        mask_file = img_file.replace('.jpg', '.png')
        
        # Copy image
        shutil.copy(
            os.path.join(raw_images, img_file),
            f"my_dataset/{split}/images/{img_file}"
        )
        
        # Copy mask
        shutil.copy(
            os.path.join(raw_masks, mask_file),
            f"my_dataset/{split}/masks/{mask_file}"
        )

print(f"Split complete:")
print(f"  Train: {len(train_files)} images")
print(f"  Valid: {len(valid_files)} images")
print(f"  Test: {len(test_files)} images")
```

Run it:

```bash
python split_dataset.py
```

### Step 3: Convert to COCO JSON

Use the conversion script:

**File: `convert_my_data.py`**

```python
import numpy as np
from pycocotools import mask as mask_utils
import json
from PIL import Image
import os
from tqdm import tqdm

def create_coco_json(images_dir, masks_dir, output_json, class_mapping):
    """
    Convert your mask images to COCO JSON
    """
    coco_data = {
        "images": [],
        "annotations": [],
        "categories": []
    }
    
    # Create categories
    for class_id, class_name in class_mapping.items():
        if class_id == 0:
            continue
        coco_data["categories"].append({
            "id": class_id,
            "name": class_name,
            "supercategory": "stuff",
            "isthing": 0  # Semantic segmentation
        })
    
    annotation_id = 1
    image_files = sorted([f for f in os.listdir(images_dir) 
                         if f.endswith(('.jpg', '.png', '.jpeg'))])
    
    print(f"Processing {len(image_files)} images...")
    
    for img_id, img_filename in enumerate(tqdm(image_files), start=1):
        # Load image
        img = Image.open(os.path.join(images_dir, img_filename))
        width, height = img.size
        
        # Add image info
        coco_data["images"].append({
            "id": img_id,
            "file_name": img_filename,
            "height": height,
            "width": width
        })
        
        # Load mask
        mask_filename = img_filename.replace('.jpg', '.png').replace('.jpeg', '.png')
        mask_img = np.array(Image.open(os.path.join(masks_dir, mask_filename)))
        
        # Process each class
        for class_id, class_name in class_mapping.items():
            if class_id == 0:
                continue
            
            # Extract binary mask for this class
            binary_mask = (mask_img == class_id).astype(np.uint8)
            
            if not np.any(binary_mask):
                continue
            
            # Convert to RLE
            rle = mask_utils.encode(np.asfortranarray(binary_mask))
            rle["counts"] = rle["counts"].decode("utf-8")
            
            # Calculate bbox
            y_indices, x_indices = np.where(binary_mask > 0)
            x_min, x_max = int(x_indices.min()), int(x_indices.max())
            y_min, y_max = int(y_indices.min()), int(y_indices.max())
            bbox = [x_min, y_min, x_max - x_min + 1, y_max - y_min + 1]
            
            # Add annotation
            coco_data["annotations"].append({
                "id": annotation_id,
                "image_id": img_id,
                "category_id": class_id,
                "bbox": bbox,
                "area": int(np.sum(binary_mask)),
                "segmentation": rle,
                "iscrowd": 1  # Semantic segmentation
            })
            annotation_id += 1
    
    # Save JSON
    with open(output_json, 'w') as f:
        json.dump(coco_data, f)
    
    print(f"\n✓ Created: {output_json}")
    print(f"  - Images: {len(coco_data['images'])}")
    print(f"  - Annotations: {len(coco_data['annotations'])}")

# Configuration
class_mapping = {
    0: "ignore",     # Skip
    1: "sand",
    2: "air",
    3: "background"
}

# Convert each split
for split in ['train', 'valid', 'test']:
    create_coco_json(
        images_dir=f"my_dataset/{split}/images",
        masks_dir=f"my_dataset/{split}/masks",
        output_json=f"my_dataset/{split}/_annotations.coco.json",
        class_mapping=class_mapping
    )
```

## 🔧 Quick Troubleshooting

```yaml

paths:
  bpe_path: /full/path/to/sam3/assets/bpe_simple_vocab_16e6.txt.gz
```

### Issue: "Out of memory"

```yaml
# Reduce batch size:
scratch:
  train_batch_size: 1  # Try 1 instead of 2
```

### Issue: "Data loader error"

```bash
# Check your JSON was created correctly:
python -c "import json; print(json.load(open('my_dataset/train/_annotations.coco.json'))['images'][0])"
```
